{{- /* Search override: supports chunked Pagefind search (default for LeapMux) while keeping Hextra's command palette UI. */ -}} {{- if (site.Params.search.enable | default true) -}} {{- $searchType := site.Params.search.type | default "pagefind" -}} {{- if eq $searchType "pagefind" -}} {{- $searchSrc := resources.Get "js/pagefind-search.js" -}} {{- if not $searchSrc -}} {{- errorf "scripts/search.html: missing resource js/pagefind-search.js" -}} {{- end -}} {{- $dialogSrc := resources.Get "js/search-dialog.js" -}} {{- if not $dialogSrc -}} {{- errorf "scripts/search.html: missing resource js/search-dialog.js" -}} {{- end -}} {{- $jsSearch := $searchSrc | resources.ExecuteAsTemplate (printf "%s.pagefind-search.js" .Language.Lang) . -}} {{- $jsDialog := $dialogSrc | resources.ExecuteAsTemplate (printf "%s.search-dialog.js" .Language.Lang) . -}} {{- if hugo.IsProduction -}} {{- $jsSearch = $jsSearch | minify | fingerprint -}} {{- $jsDialog = $jsDialog | minify | fingerprint -}} {{- end -}} {{- else -}} {{- errorf `scripts/search.html: unsupported search type %q (expected "pagefind")` $searchType -}} {{- end -}} {{- end -}}
